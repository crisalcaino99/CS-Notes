\section{Sobre la funcion gamma}
\begin{Definition}{La funcion gamma}
    -- Definimos la funcion gamma de la siguiente forma:
    $$ z \in \complex : \Gamma(z) := \int_{0}^{\infty} t^{z-1}e^{-t} dt $$
\end{Definition}

\begin{Proposition}{Una sustitucion silenciosa}
    -- $$ t^{n} = e^{\ln(t)n} $$
\end{Proposition}

\begin{Proposition}{La exponencial supera a los polinomios}
    -- $$ \forall n \in \naturals \exists C := (2n)! : t^{n} \leq C e^{\frac{t}{2}}$$
\end{Proposition}

\begin{Proposition}{La integral converge}
    -- Las siguientes integrales convergen
    $$ \int_{0}^{\infty} t^{n} e^{-\frac{t}{2}} dt $$
    $$ \int_{0}^{\infty} e^{-t} \frac{1}{\sqrt{t}} dt $$
\end{Proposition}

\begin{Proposition}{Gamma esta bien definida}
    -- $$ z \in \complex: Re(z) > 0 \then \Gamma(z) : Re(\Gamma(z)) = \int_{0}^{\infty} t^{u-1} e^{-t} \cos(v \ln(t)) dt $$
\end{Proposition}
