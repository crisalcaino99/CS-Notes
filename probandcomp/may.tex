\section{Algunos comentarios sobre cotas}

\begin{Proposition}
    -- $$ \lim_{k \to \infty} \binom{2k}{k} \Big( \frac{1}{2} \Big)^{2k}  \leq 1 $$ 
\end{Proposition}

Puede ser que tengamos proposiciones similares sobre cotas como por ejemplo

\begin{Proposition}
    -- $$ \binom{2k}{k} \leq 4^k$$ 
\end{Proposition}

Conviene tener presente:
\begin{Theorem}{Binomio de newton}
    - $$a, b \in \reals, n \in \naturals \then (a+b)^n = \sum_{j=0}^n \binom{n}{k} a^{n-j}b^j$$
\end{Theorem}

\begin{Proposition}{Stirling}
    - Por demostrar:
    - $$n \in \naturals \then n! \sim \sqrt{2\pi n } \Big( \frac{n}{e} \Big)^n $$

\end{Proposition}
