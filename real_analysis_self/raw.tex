\section{Algunas notas sobre análisis}

// Lema de urysohn
\begin{Definition}{Espacio normal}
    -- Decimos que X es normal si:
    $$ \forall A, B \subseteq X: A = \bar{A}, B = \bar{B} \then \exists U, V opens: A 
    \subseteq U, B \subseteq V, U \cap V = \emptyset $$
\end{Definition}

\begin{Lemma}{Lema de Urysohn}
    -- Sea X normal, A, B $\subseteq$ cerrados disjuntos 
    $$\then \exists f : f: X \to [0, 1] \text{continua} $$
    tal que 
    $$ f|_A = 0, f|_B = 1 $$ 
\end{Lemma}


\begin{Theorem}{Stone - Weierstrass}
    -- $A \subseteq \mathcal{C}(X, \reals)_{|.|_\infty}$
    tal que A contiene a las constantes y para cualquier funcion $f \in A$ separa puntos
    $\then \bar{A}^{||.||_\infty} = \mathcal{C}(X, \reals)$ 
\end{Theorem}

-- Compactificacion de la metrica

\begin{Proposition}{Compactificacion de la metrica}
    -- Sea $X$ un espacio metrico con $d$ su metrica. Entonces 
    se tiene que si $f$ es creciente: $f(x) = 0 \iff x = 0$ y subaditiva ($f(x + y) \leq f(x) + f(y)$)
    entonces $f_d(x, y) = f(d(x, y))$ es metrica. 
\end{Proposition}

\begin{Proposition}{Bola unitaria, continuas y supremo}
    -- $B_1$ en $\mathcal{C}([a, b])$ con $||.||_\infty$ 
    no es compacto.
\end{Proposition}
